/* =========================
LOCAL STORAGE
========================= */
function getArticles() {
const saved = localStorage.getItem("articles");
if (saved) {
return JSON.parse(saved);
}
localStorage.setItem(
"articles",
JSON.stringify(defaultArticles)
);
return defaultArticles;
}
function saveArticles(articles) {
localStorage.setItem(
"articles",
JSON.stringify(articles)
);
}
/* =========================
TAMPILKAN ARTIKEL
========================= */
function displayArticles(list = getArticles()) {
const container = document.getElementById("articleList");
if (!container) return;
container.innerHTML = "";
if (list.length === 0) {
container.innerHTML =
"<p>Tidak ada artikel ditemukan.</p>";
return;
}
list.forEach(article => {
const card = document.createElement("div");
card.className = "article-card";
card.innerHTML = `
<img
src="${article.image || "images/foto1.jpg"}"
alt="${article.title}"
>
<div class="article-info">
<small>${article.category}</small>
<h3>${article.title}</h3>
<p>
${article.content.substring(0, 100)}...
</p>
<br>
<a
class="btn"
href="artikel.html?id=${article.id}"
>
Baca Selengkapnya
</a>
</div>
`;
container.appendChild(card);
});
}
/* =========================
PENCARIAN
========================= */
const searchInput =
document.getElementById("searchInput");
if (searchInput) {
searchInput.addEventListener("input", function () {
const keyword =
this.value.toLowerCase();
const filtered =
getArticles().filter(article =>
article.title
.toLowerCase()
.includes(keyword)
||
article.category
.toLowerCase()
.includes(keyword)
||
article.content
.toLowerCase()
.includes(keyword)
);
displayArticles(filtered);
});
}
/* =========================
DETAIL ARTIKEL
========================= */
function showArticleDetail() {
const container =
document.getElementById("articleDetail");
if (!container) return;
const params =
new URLSearchParams(window.location.search);
const id =
Number(params.get("id"));
const article =
getArticles().find(item => item.id === id);
if (!article) {
container.innerHTML = `
<h1>Artikel tidak ditemukan</h1>
<br>
<a class="btn" href="index.html">
Kembali
</a>
`;
return;
}
const paragraphs =
article.content
.split("\n")
.map(text => `<p>${text}</p>`)
.join("");
container.innerHTML = `
<small class="small-title">
${article.category}
</small>
<h1>${article.title}</h1>
<p>
Ditulis oleh <strong>${article.author}</strong>
</p>
<img
src="${article.image || "images/foto1.jpg"}"
alt="${article.title}"
>
<div class="article-content">
${paragraphs}
</div>
<br>
<a class="btn" href="index.html">
← Kembali
</a>
`;
}
/* =========================
ADMIN
========================= */
const articleForm =
document.getElementById("articleForm");
if (articleForm) {
articleForm.addEventListener(
"submit",
function(event) {
event.preventDefault();
const id =
document.getElementById("editId").value;
const article = {
id: id
? Number(id)
: Date.now(),
title:
document.getElementById("title").value,
category:
document.getElementById("category").value,
author:
document.getElementById("author").value,
image:
document.getElementById("image").value
|| "images/foto1.jpg",
content:
document.getElementById("content").value
};
let articles = getArticles();
if (id) {
articles =
articles.map(item =>
item.id === Number(id)
? article
: item
);
} else {
articles.unshift(article);
}
saveArticles(articles);
resetForm();
displayAdminArticles();
alert("Artikel berhasil disimpan!");
}
);
}
/* =========================
ADMIN LIST
========================= */
function displayAdminArticles() {
const container =
document.getElementById("adminArticles");
if (!container) return;
container.innerHTML = "";
getArticles().forEach(article => {
const item =
document.createElement("div");
item.className = "admin-item";
item.innerHTML = `
<div>
<strong>${article.title}</strong>
<br>
<small>
${article.category}
— ${article.author}
</small>
</div>
<div class="admin-actions">
<button
class="edit-btn"
onclick="editArticle(${article.id})"
>
Edit
</button>
<button
class="delete-btn"
onclick="deleteArticle(${article.id})"
>
Hapus
</button>
</div>
`;
container.appendChild(item);
});
}
/* =========================
EDIT ARTIKEL
========================= */
function editArticle(id) {
const article =
getArticles().find(item => item.id === id);
if (!article) return;
document.getElementById("editId").value =
article.id;
document.getElementById("title").value =
article.title;
document.getElementById("category").value =
article.category;
document.getElementById("author").value =
article.author;
document.getElementById("image").value =
article.image;
document.getElementById("content").value =
article.content;
window.scrollTo({
top: 0,
behavior: "smooth"
});
}
/* =========================
HAPUS ARTIKEL
========================= */
function deleteArticle(id) {
if (!confirm("Hapus artikel ini?")) {
return;
}
const articles =
getArticles().filter(item => item.id !== id);
saveArticles(articles);
displayAdminArticles();
displayArticles();
}
/* =========================
RESET FORM
========================= */
function resetForm() {
const form =
document.getElementById("articleForm");
if (!form) return;
form.reset();
document.getElementById("editId").value = "";
}
/* =========================
DARK MODE
========================= */
const darkButton =
document.getElementById("darkMode");
if (darkButton) {
const dark =
localStorage.getItem("darkMode");
if (dark === "true") {
document.body.classList.add("dark");
}
darkButton.addEventListener("click", () => {
document.body.classList.toggle("dark");
localStorage.setItem(
"darkMode",
document.body.classList.contains("dark")
);
});
}
